텍스트는 `TextWidget`으로 형변환해야 `SetTextFormat`을 쓸 수 있지만, 보이고 숨기는 `SetVisible`은 모든 위젯의 부모인 `Widget`에 있으므로 이미지 위젯은 형변환하지 않아도 됩니다.

```enforce
void UpdateScore(Widget root, int score)
{
	if (!root)
		return;

	bool negative = score < 0;

	TextWidget scoreText = TextWidget.Cast(root.FindAnyWidget("ScoreText"));
	if (scoreText)
	{
		scoreText.SetTextFormat("점수: %1", score);
		if (negative)
			scoreText.SetColor(Color.Red);
		else
			scoreText.SetColor(Color.White);
	}

	Widget warningIcon = root.FindAnyWidget("WarningIcon");
	if (warningIcon)
		warningIcon.SetVisible(negative);
}
```

위젯을 이름으로 찾은 결과는 이름이 틀렸거나 레이아웃이 바뀌면 null이 될 수 있으므로 하나씩 확인했습니다. 이 메서드를 자주 부른다면 위젯을 매번 찾지 말고 처음 한 번 찾아 `m_w...` 멤버에 저장해 두는 편이 좋습니다.

<Answer id="13-2">

이름 부분은 `<color>` 안에 `<b>`를 넣고, 닫을 때는 연 순서의 반대로 닫습니다. 스크립트 문자열 안이므로 속성 값은 공식 코드처럼 작은따옴표로 감쌌습니다.

```enforce
string TAG_MakeWelcomeText(string playerName)
{
	string nameText = string.Format("<color name='green'><b>%1</b></color>", playerName);
	return nameText + "님, 환영합니다<br/><action name='CharacterAction'/> 키를 눌러 시작하세요";
}
```

- `<br/>`가 줄을 바꿉니다.
- `<action name='CharacterAction'/>`은 그 플레이어의 키 설정에 맞는 키 아이콘으로 바뀝니다. 아이콘 대신 글자로 보이게 하려면 `mode='text'`를 더합니다.
- 이 문자열은 `RichTextWidget`에 넣어야 태그가 해석됩니다. RichText Format 문서에 따르면 굵은 글씨(`<b>`)는 SDF 글꼴에서만 지원됩니다.
- 실제 모드에서는 "님, 환영합니다" 같은 문장도 로컬라이즈 키로 바꾸는 것이 좋습니다.

</Answer>

<Answer id="13-3">

공식 문서는 `HandlerAttached`가 불릴 때 레이아웃의 나머지 위젯이 모두 만들어져 있는지 설명하지 않습니다. 공식 `SCR_ButtonTextComponent`는 `HandlerAttached`에서 **자기 위젯의 자식**을 찾지만, 이 문제의 대상은 레이아웃 어디에든 있을 수 있습니다. 그래서 이 해답은 실제로 필요한 순간, 즉 처음 클릭할 때 찾고 결과를 저장해 둡니다. 찾을 때는 버튼에서 부모 쪽으로 한 단계씩 올라가며 `FindAnyWidget`으로 검색하므로 가장 가까운 범위의 위젯이 먼저 찾아집니다.

```enforce title="Scripts/Game/UI/TAG_ToggleVisibilityComponent.c"
class TAG_ToggleVisibilityComponent : ScriptedWidgetComponent
{
	[Attribute(desc: "숨기거나 보일 위젯 이름")]
	protected string m_sTargetName;

	protected Widget m_wButton;
	protected Widget m_wTarget;

	override void HandlerAttached(Widget w)
	{
		super.HandlerAttached(w);
		m_wButton = w;
	}

	override bool OnClick(Widget w, int x, int y, int button)
	{
		if (!m_wTarget)
			m_wTarget = FindTarget();

		if (!m_wTarget)
			return false;	// 대상이 없으면 처리하지 않은 것으로 둔다

		m_wTarget.SetVisible(!m_wTarget.IsVisible());
		return true;
	}

	protected Widget FindTarget()
	{
		Widget parent = m_wButton;
		while (parent)
		{
			Widget found = parent.FindAnyWidget(m_sTargetName);
			if (found)
				return found;

			parent = parent.GetParent();
		}

		return null;
	}
}
```

- `m_wTarget`은 약한 참조이므로 대상 위젯이 지워지면 null이 되고, 다음 클릭 때 다시 찾습니다.
- 대상을 찾지 못하면 `false`를 돌려줍니다. 공식 주석에 따르면 처리되지 않은 이벤트는 부모 위젯 쪽으로 계속 전달됩니다.

</Answer>

<Answer id="13-4">

입력 리스너를 등록해 두었다가 액션이 들어오면 메뉴를 엽니다. 이미 열려 있는지는 `MenuManager.FindMenuByPreset()`으로 확인합니다. `OpenMenu()`의 세 번째 인수 `unique`를 true로 주면 이미 열린 같은 메뉴를 돌려주므로 그 방법도 됩니다.

```enforce title="Scripts/Game/TAG_InfoMenuOpenerComponent.c"
[ComponentEditorProps(category: "TAG/UI", description: "키를 누르면 정보 메뉴를 엽니다")]
class TAG_InfoMenuOpenerComponentClass : ScriptComponentClass
{
}

class TAG_InfoMenuOpenerComponent : ScriptComponent
{
	protected void OpenInfoMenu()
	{
		MenuManager menuManager = GetGame().GetMenuManager();
		if (!menuManager || menuManager.FindMenuByPreset(ChimeraMenuPreset.TAG_InfoMenu))
			return;

		menuManager.OpenMenu(ChimeraMenuPreset.TAG_InfoMenu);
	}

	override void OnPostInit(IEntity owner)
	{
		InputManager inputManager = GetGame().GetInputManager();
		if (inputManager && !SCR_Global.IsEditMode(owner))
			inputManager.AddActionListener("TAG_OpenInfo", EActionTrigger.DOWN, OpenInfoMenu);
	}

	override void OnDelete(IEntity owner)
	{
		InputManager inputManager = GetGame().GetInputManager();
		if (inputManager)
			inputManager.RemoveActionListener("TAG_OpenInfo", EActionTrigger.DOWN, OpenInfoMenu);
	}
}
```

- 등록은 `OnPostInit`, 해제는 `OnDelete`에서 합니다. World Editor 편집 중에는 등록하지 않습니다(10장).
- 콜백에 매개변수가 없어도 됩니다. 공식 코드에도 매개변수 없는 액션 리스너가 많습니다.
- 리스너가 불리려면 `TAG_OpenInfo` 액션이 입력 설정에 정의되어 있고, 그 액션이 속한 컨텍스트가 켜져 있어야 합니다(11장).
- 공식 `game.c`는 메뉴가 하나도 열려 있지 않을 때만 매 프레임 `IngameContext`를 켭니다. 그래서 `TAG_OpenInfo`를 `IngameContext`에 넣었다면 메뉴가 열린 동안에는 이 액션이 들어오지 않습니다. 메뉴를 닫는 입력은 본문처럼 메뉴 클래스에서 처리합니다.

</Answer>

<Answer id="13-5">

1초마다 반복하는 `Tick()`이 남은 시간을 줄이다가 0이 되면 반복을 멈추고 "시작!"을 보여 준 뒤, 2초 뒤 `Stop()`을 예약합니다. `Stop()`은 두 예약을 모두 지우고 레이아웃도 지우므로 언제 불러도 안전합니다.

```enforce title="Scripts/Game/UI/TAG_CountdownDisplay.c"
class TAG_CountdownDisplay
{
	protected Widget m_wRoot;
	protected TextWidget m_wMessage;
	protected int m_iRemaining;

	void Start(ResourceName layout, int seconds)
	{
		Stop();		// 이전 카운트다운이 있으면 정리

		m_wRoot = GetGame().GetWorkspace().CreateWidgets(layout);
		if (!m_wRoot)
			return;

		m_wMessage = TextWidget.Cast(m_wRoot.FindAnyWidget("Message"));
		m_iRemaining = seconds;
		Tick();
		GetGame().GetCallqueue().CallLater(Tick, 1000, true);
	}

	void Stop()
	{
		ScriptCallQueue queue = GetGame().GetCallqueue();
		queue.Remove(Tick);
		queue.Remove(Stop);

		if (m_wRoot)
			m_wRoot.RemoveFromHierarchy();

		m_wRoot = null;
		m_wMessage = null;
	}

	// Tick()과 소멸자는 아래에 이어진다
}
```

```enforce
protected void Tick()
{
	if (m_iRemaining > 0)
	{
		if (m_wMessage)
			m_wMessage.SetTextFormat("남은 시간: %1초", m_iRemaining);

		m_iRemaining--;
		return;
	}

	ScriptCallQueue queue = GetGame().GetCallqueue();
	queue.Remove(Tick);
	if (m_wMessage)
		m_wMessage.SetText("시작!");

	queue.CallLater(Stop, 2000);
}

void ~TAG_CountdownDisplay()
{
	Stop();
}
```

- `Start()`에서 `Tick()`을 직접 한 번 불러 첫 숫자를 바로 보여 줍니다.
- 반복 호출 안에서 자기 자신을 `Remove`하는 방식은 `ScriptCallQueue`의 공식 주석 예제와 같습니다.
- 소멸자에서도 `Stop()`을 불러 객체가 사라진 뒤 큐에 예약이 남지 않게 합니다. 이 객체를 만든 쪽은 카운트다운이 끝날 때까지 `ref`로 들고 있어야 합니다(7장).

</Answer>
